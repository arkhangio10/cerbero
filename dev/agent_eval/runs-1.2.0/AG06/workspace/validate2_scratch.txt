def parse(t,recs,errs):
    lines=t.split('\n');cur=None;hdr={}
    for ln in lines[1:]:
        if not ln: continue
        m=re.match(r'^## RECORD (\S+)$',ln)
        if m:
            cur=m.group(1)
            if cur in recs: errs.append('dup '+cur)
            recs[cur]={};continue
        m=re.match(r'^- ([a-z0-9_]+): (.*)$',ln)
        if not m: errs.append('bad line '+ln);continue
        try: v=json.loads(m.group(2))
        except Exception as e: errs.append('json '+ln);continue
        tgt=hdr if cur is None else recs[cur]
        if m.group(1) in tgt: errs.append('dup field '+ln)
        tgt[m.group(1)]=v
    return hdr
